    <script>
        // Apply gradient to all headings as soon as possible
        document.addEventListener('DOMContentLoaded', () => {
            const headings = document.querySelectorAll('h1, h2, h3, h4, h5, h6');
            headings.forEach(h => {
                // Skip the main hero text as it already has custom styling and white text
                if (h.closest('.fade-in-left') && h.tagName === 'H1') return;
                
                // Skip the new 3D slider cards to keep their white/orange colors
                if (h.classList.contains('card-title')) return;
                if (h.classList.contains('skip-gradient')) return;
                
                // Use a lighter gradient for footer headings so they are visible on dark backgrounds
                if (h.closest('footer') || h.closest('.bg-black')) {
                    h.classList.add('gradient-heading-light');
                } else {
                    h.classList.add('gradient-heading');
                }
            });
        });
    </script>
