        function checkVisibility() {
            const elements = document.querySelectorAll('.animate-on-scroll');
            const windowHeight = window.innerHeight;

            elements.forEach(element => {
                const elementTop = element.getBoundingClientRect().top;
                const elementBottom = element.getBoundingClientRect().bottom;

                if (elementTop < windowHeight * 0.85 && elementBottom > 0) {
                    element.classList.add('visible');
                }
            });
        }

        window.addEventListener('scroll', checkVisibility);
        window.addEventListener('load', checkVisibility);

        checkVisibility();

    <script>
        document.addEventListener("DOMContentLoaded", () => {
            const fadeElements = document.querySelectorAll(
                ".fade-in-up, .fade-in-left, .fade-in-right, .scale-in"
            );

            const observer = new IntersectionObserver(
                (entries) => {
                    entries.forEach((entry) => {
                        if (entry.isIntersecting) {
                            // Element is in view
                            entry.target.classList.add("active");
                        } else {
                            // Element went out of view — remove the class so it reanimates
                            entry.target.classList.remove("active");
                        }
                    });
                },
                {
                    threshold: 0.2, // Adjust how early it triggers
                }
            );

            fadeElements.forEach((el) => observer.observe(el));
        });
    </script>


    <script>
        const footerContainer = document.getElementById('footer-container');
        const cursor = document.getElementById('custom-cursor');
        const cursorTrail = document.getElementById('cursor-trail');

        let mouseX = 0;
        let mouseY = 0;
        let cursorX = 0;
        let cursorY = 0;
        let trailX = 0;
        let trailY = 0;

        footerContainer.addEventListener('mouseenter', () => {
            footerContainer.classList.add('cursor-active');
        });

        footerContainer.addEventListener('mouseleave', () => {
            footerContainer.classList.remove('cursor-active');
        });

        footerContainer.addEventListener('mousemove', (e) => {
            mouseX = e.clientX;
            mouseY = e.clientY;
        });

        footerContainer.addEventListener('mousedown', () => {
            footerContainer.classList.add('clicking');
        });

        footerContainer.addEventListener('mouseup', () => {
            footerContainer.classList.remove('clicking');
        });

        function animateCursor() {
            const speed = 0.15;
            const trailSpeed = 0.1;

            cursorX += (mouseX - cursorX) * speed;
            cursorY += (mouseY - cursorY) * speed;

            trailX += (mouseX - trailX) * trailSpeed;
            trailY += (mouseY - trailY) * trailSpeed;

            cursor.style.transform = `translate(${cursorX - 10}px, ${cursorY - 10}px)`;
            cursorTrail.style.transform = `translate(${trailX - 20}px, ${trailY - 20}px)`;

            requestAnimationFrame(animateCursor);
        }

        animateCursor();

        const hoverableText = footerContainer.querySelectorAll('.hoverable-text, .hoverable-link');
        const hoverableButtons = footerContainer.querySelectorAll('.hoverable-button, .hoverable-input');

        hoverableText.forEach(element => {
            element.addEventListener('mouseenter', () => {
                footerContainer.classList.add('hovering-text');
            });
            element.addEventListener('mouseleave', () => {
                footerContainer.classList.remove('hovering-text');
            });
        });

        hoverableButtons.forEach(element => {
            element.addEventListener('mouseenter', () => {
                footerContainer.classList.add('hovering-clickable');
            });
            element.addEventListener('mouseleave', () => {
                footerContainer.classList.remove('hovering-clickable');
            });
        });
    </script>

    <script>
        const words = ['Websites', 'E-Commerce'];
        let wordIndex = 0;
        let charIndex = 0;
        let isDeleting = false;
        const typingSpeed = 150;
        const deletingSpeed = 50;
        const pauseTime = 2000;

        function typeEffect() {
            const typingElement = document.getElementById('typingText');
            const currentWord = words[wordIndex];

            if (!isDeleting) {
                typingElement.textContent = currentWord.substring(0, charIndex);
                charIndex++;

                if (charIndex > currentWord.length) {
                    isDeleting = true;
                    setTimeout(typeEffect, pauseTime);
                    return;
                }
            } else {
                typingElement.textContent = currentWord.substring(0, charIndex);
                charIndex--;

                if (charIndex === 0) {
                    isDeleting = false;
                    wordIndex = (wordIndex + 1) % words.length;
                }
            }

            setTimeout(typeEffect, isDeleting ? deletingSpeed : typingSpeed);
        }

        const observerOptions = {
            threshold: 0.1,
            rootMargin: '0px 0px -100px 0px'
        };

        const observer = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    entry.target.classList.add('active');
                }
            });
        }, observerOptions);

        document.addEventListener('DOMContentLoaded', () => {
            const animatedElements = document.querySelectorAll('.fade-in-up, .fade-in-left, .fade-in-right, .scale-in');
            animatedElements.forEach(el => observer.observe(el));

            typeEffect();
        });
    </script>

    <script>
        let currentIndex = 1;
        let isAnimating = false;
        const totalSlides = 5;

        function updateSlides() {
            for (let i = 0; i < totalSlides; i++) {
                const slide = document.getElementById(`slide-${i}`);
                const slideContent = slide.querySelector('.slide-content');
                const compactView = slide.querySelector('.compact-view');
                const expandedView = slide.querySelector('.expanded-view');
                const dot = document.getElementById(`dot-${i}`);

                slide.classList.remove('slide-center', 'slide-left', 'slide-right', 'slide-hidden');

                if (i === currentIndex) {
                    dot.classList.remove('w-2', 'bg-gray-400');
                    dot.classList.add('w-8', 'bg-emerald-400');
                } else {
                    dot.classList.remove('w-8', 'bg-emerald-400');
                    dot.classList.add('w-2', 'bg-gray-400');
                }

                const diff = (i - currentIndex + totalSlides) % totalSlides;
                let position;

                if (diff === 0) {
                    position = 'center';
                    slideContent.classList.remove('compact-card');
                    slideContent.classList.add('expanded-card');
                    compactView.classList.add('hidden');
                    expandedView.classList.remove('hidden');
                } else if (diff === totalSlides - 1) {
                    position = 'left';
                    slideContent.classList.remove('expanded-card');
                    slideContent.classList.add('compact-card');
                    compactView.classList.remove('hidden');
                    expandedView.classList.add('hidden');
                } else if (diff === 1) {
                    position = 'right';
                    slideContent.classList.remove('expanded-card');
                    slideContent.classList.add('compact-card');
                    compactView.classList.remove('hidden');
                    expandedView.classList.add('hidden');
                } else {
                    position = 'hidden';
                    slideContent.classList.remove('expanded-card');
                    slideContent.classList.add('compact-card');
                    compactView.classList.remove('hidden');
                    expandedView.classList.add('hidden');
                }

                slide.classList.add(`slide-${position}`);
            }
        }

        function prevSlide() {
            if (isAnimating) return;
            isAnimating = true;
            currentIndex = currentIndex === 0 ? totalSlides - 1 : currentIndex - 1;
            updateSlides();
            setTimeout(() => { isAnimating = false; }, 600);
        }

        function nextSlide() {
            if (isAnimating) return;
            isAnimating = true;
            currentIndex = currentIndex === totalSlides - 1 ? 0 : currentIndex + 1;
            updateSlides();
            setTimeout(() => { isAnimating = false; }, 600);
        }

        function goToSlide(index) {
            if (isAnimating || index === currentIndex) return;
            isAnimating = true;
            currentIndex = index;
            updateSlides();
            setTimeout(() => { isAnimating = false; }, 600);
        }

        updateSlides();

            // 🔹 Initially hidden glow
            glow.style.opacity = "0";

            // 🔹 Mouse enters → fade in + reset to top-left
            card.addEventListener("mouseenter", () => {
                glow.style.opacity = "1";
                glow.style.background = `radial-gradient(circle at 20px 20px, rgba(64, 196, 155, 1) 0%, transparent 30%)`;
            });

            // 🔹 Move glow horizontally along the top while hovering
            card.addEventListener("mousemove", (e) => {
                const rect = card.getBoundingClientRect();
                const x = e.clientX - rect.left; // Mouse X inside card
                const y = 20; // Fixed top position for glow
                glow.style.background = `radial-gradient(circle at ${x}px ${y}px, rgba(64, 196, 155, 1) 0%, transparent 30%)`;
            });

            // 🔹 Mouse leaves → fade out + reset back to top-left
            card.addEventListener("mouseleave", () => {
                glow.style.opacity = "0";
                glow.style.transition = "opacity 0.4s ease, background 0.4s ease-out";
                glow.style.background = `radial-gradient(circle at 20px 20px, rgba(64, 196, 155, 1) 0%, transparent 30%)`;
            });
        });

    <script>
        // Sticky Navbar logic
        window.addEventListener('scroll', () => {
            const headerInner = document.getElementById('header-inner');
            if (!headerInner) return;
            if (window.scrollY > 50) {
                headerInner.classList.add('fixed', 'top-2', 'z-[100]', 'bg-white/80', 'backdrop-blur-md', 'left-0', 'right-0', 'mx-auto');
            } else {
                headerInner.classList.remove('fixed', 'top-2', 'z-[100]', 'bg-white/80', 'backdrop-blur-md', 'left-0', 'right-0', 'mx-auto');
            }
        });

        // Slider Auto-play logic
        let sliderInterval = setInterval(nextSlide, 3500);

        // Pause slider on hover
        const sliderContainer = document.getElementById('slider-container');
        if (sliderContainer) {
            sliderContainer.addEventListener('mouseenter', () => clearInterval(sliderInterval));
            sliderContainer.addEventListener('mouseleave', () => {
                sliderInterval = setInterval(nextSlide, 3500);
            });
        }
    </script>
